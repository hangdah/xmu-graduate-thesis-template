% ==================== 致谢标题、页眉页脚与目录条目 ====================

%%%%%%%% 致谢 %%%%%%%%

% 在致谢正文中调用；第二个参数填写日期及地点，不写死作者信息。
\newcommand{\acknowledgementsignature}[2]{%
    \par\vspace{\baselineskip}%
    \begingroup\raggedleft\setlength{\parindent}{0pt}%
        #1\par
        #2\par
    \endgroup
}

\newenvironment{acknowledgement}{
    \chapter*{致\quad\; 谢}
    \markboth{致谢}{致谢}%
    \pagestyle{fancy}
    \fancyhf{}
    \ifthenelse{\isundefined{\xmu@entoc}}{
        % 如果致谢在目录前，使用前置页面样式，但不添加目录项。
        \pagestyle{xmu@front}\thispagestyle{xmu@front}
    }{
        % 如果致谢在目录后，设置页眉、页脚，并添加索引到目录
        \renewcommand{\headrulewidth}{0.4pt}
        \fancyhead[OC]{\rm\zihao{-5}致\quad\; 谢}
        \fancyhead[EC]{\rm\zihao{-5}\xmu@title}
        \fancyfoot[C]{\rm\zihao{-5}\thepage}
        \addcontentsline{toc}{chapter}{致谢}
        \addcontentsline{entoc}{chapter}{Acknowledgements}
    }
}{
    \par
    \clearpage
}
